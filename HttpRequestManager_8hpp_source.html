250 void process_pending_requests() {
251 std::unique_lock<std::mutex> lock(m_mutex);
252 if (m_pending_requests.empty()) return;
253
254 std::vector<std::unique_ptr<HttpRequestContext>> pending_request;
255 std::vector<std::unique_ptr<HttpRequestContext>> failed_requests;
256
257 auto it = m_pending_requests.begin();
258 while (it != m_pending_requests.end()) {
259 auto& context = *it;
260 auto& request = context->request;
261 // Check if the request is valid.
262 if (!request) {
263 failed_requests.push_back(std::move(context));
264 it = m_pending_requests.erase(it);
265 continue;
266 }
267
268 // Set up completion callback before allow_request so it is armed
269 // even if an exception occurs after the token is committed.
270 auto general_limit = request->general_rate_limit;
271 auto specific_limit = request->specific_rate_limit;
272 uint64_t token = context->in_flight_token;
273 auto general_key = request->general_rate_limit_key;
274 auto specific_key = request->specific_rate_limit_key;
275
276 // Preserve any previous on_complete so a retry that already owns
277 // a sequential token does not lose its cleanup callback.
278 auto old_on_complete = std::move(context->on_complete);
279 context->on_complete = [this, general_limit, specific_limit, token, general_key, specific_key]() {
280 m_rate_limiter.release_request(general_limit, specific_limit, token, general_key, specific_key);
281 };
282
283 // Check if the request is allowed by the rate limiter.
284 const bool allowed = m_rate_limiter.allow_request(
285 general_limit,
286 specific_limit,
287 token,
288 request->general_rate_limit_key,
289 request->specific_rate_limit_key);
290 if (!allowed) {
291 context->on_complete = std::move(old_on_complete);
292 ++it;
293 continue;
294 }
295
296 pending_request.push_back(std::move(context));
297 it = m_pending_requests.erase(it);
298 }
299 lock.unlock();
300
301 // Handle failed requests by calling their callback with a 400 status.
302 if (!failed_requests.empty()) {
303 for (const auto &context : failed_requests) {
304# if __cplusplus >= 201402L
305 auto response = std::make_unique<HttpResponse>();
306# else
307 auto response = std::unique_ptr<HttpResponse>(new HttpResponse());
308# endif
309 const long BAD_REQUEST = 400;
310 response->error_code = utils::make_error_code(CURLE_OK);
311 response->status_code = BAD_REQUEST;
312 response->ready = true;
313 context->callback(std::move(response));
314 context->complete();
315 }
316 failed_requests.clear();
317 }
318
319 // If there are ready requests, create a new HttpBatchRequestHandler to manage them.
320 if (pending_request.empty()) return;
321# if __cplusplus >= 201402L
322 m_active_request_batches.push_back(std::make_unique<HttpBatchRequestHandler>(pending_request));
323# else
324 m_active_request_batches.push_back(std::unique_ptr<HttpBatchRequestHandler>(new HttpBatchRequestHandler(pending_request)));
325# endif
326 }

329 void process_active_requests() {
330 auto it = m_active_request_batches.begin();
331 while (it != m_active_request_batches.end()) {
332 auto& request = *it;
333 if (!request->process()) {
334 ++it;
335 continue;
336 }
337 auto failed_requests = request->extract_failed_requests();
338 it = m_active_request_batches.erase(it);
339 for (auto& request : failed_requests) {
340 m_failed_requests.push_back(std::move(request));
341 }
342 }

346 void process_retry_failed_requests() {
347 auto it = m_failed_requests.begin();
348 while (it != m_failed_requests.end()) {
349 auto& request_context = *it;
350 if (!request_context || !request_context->request) {
351 it = m_failed_requests.erase(it);
352 continue;
353 }
354
355 const auto now = std::chrono::steady_clock::now();
356 const auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(now - request_context->start_time);
357 const auto& retry_delay_ms = request_context->request->retry_delay_ms;
358 if (duration.count() >= retry_delay_ms) {
359 std::unique_lock<std::mutex> lock(m_mutex);
360 m_pending_requests.push_back(std::move(request_context));
361 lock.unlock();
362 it = m_failed_requests.erase(it);
363 continue;
364 }
365 ++it;
366 }

370 void process_cancel_requests() {
371 std::unique_lock<std::mutex> lock(m_mutex);
372 if (m_requests_to_cancel_by_id.empty() && m_groups_to_cancel.empty()) return;
373
374 auto requests_to_cancel = std::move(m_requests_to_cancel_by_id);
375 auto groups_to_cancel = std::move(m_groups_to_cancel);
376 m_requests_to_cancel_by_id.clear();
377 m_groups_to_cancel.clear();
378
379 std::list<std::unique_ptr<HttpRequestContext>> canceled_pending_requests;
380 auto pending_it = m_pending_requests.begin();
381 while (pending_it != m_pending_requests.end()) {
382 const auto& ctx = *pending_it;
383 if (!matches_cancel(ctx, requests_to_cancel, groups_to_cancel)) {
384 ++pending_it;
385 continue;
386 }
387 canceled_pending_requests.push_back(std::move(*pending_it));
388 pending_it = m_pending_requests.erase(pending_it);
389 }
390 lock.unlock();
391
392 for (const auto& request_context : canceled_pending_requests) {
393 request_context->callback(make_cancelled_response());
394 request_context->complete();
395 }
396
397 auto failed_it = m_failed_requests.begin();
398 while (failed_it != m_failed_requests.end()) {
399 const auto& request_context = *failed_it;
400 if (!matches_cancel(request_context, requests_to_cancel, groups_to_cancel)) {
401 ++failed_it;
402 continue;
403 }
404 request_context->callback(make_cancelled_response());
405 request_context->complete();
406 failed_it = m_failed_requests.erase(failed_it);
407 }
408
409 for (const auto &handler : m_active_request_batches) {
410 handler->cancel_requests(requests_to_cancel, groups_to_cancel);
411 }
412
413 invoke_cancel_callbacks(requests_to_cancel);
414 invoke_cancel_callbacks(groups_to_cancel);
415 }

417 static bool matches_cancel(
418 const std::unique_ptr<HttpRequestContext>& ctx,
419 const cancel_map_t& requests_to_cancel,
420 const cancel_map_t& groups_to_cancel) {
421 if (!ctx || !ctx->request) return false;
422 const auto request_id = ctx->request->request_id;
423 const auto group_id = ctx->request->group_id;
424 return
425 (request_id != 0 && requests_to_cancel.count(request_id) > 0) ||
426 (group_id != 0 && groups_to_cancel.count(group_id) > 0);
427 }
