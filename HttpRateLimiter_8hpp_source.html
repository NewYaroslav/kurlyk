406 template<typename Duration>
407 Duration time_until_limit_allows(
408 const LimitData& limit_data,
409 const time_point_t& now
410 ) const {
411 if (limit_data.sequential &&
412 !limit_data.in_flight_tokens.empty()) {
413 return Duration::max();
414 }
415
416 if (limit_data.requests_per_period == 0) {
417 return Duration{0};
418 }
419
420 const auto elapsed =
421 std::chrono::duration_cast<Duration>(now - limit_data.start_time);
422
423 const auto period_duration =
424 std::chrono::duration_cast<Duration>(
425 std::chrono::milliseconds(limit_data.period_ms)
426 );
427
428 if (elapsed >= period_duration ||
429 limit_data.count < limit_data.requests_per_period) {
430 return Duration{0};
431 }
432
433 return period_duration - elapsed;
434 }
