402 bool can_pass(const LimitData& limit, const std::string& key, uint64_t token, const time_point_t& now) const {
403 const KeyState* state = find_key_state(limit, key);
404 if (!state) {

485 const LimitData& limit,
486 const std::string& key,
487 const time_point_t& now

489 const KeyState* state = find_key_state(limit, key);
490 if (!state) {
491 // No state means no in-flight tokens and count is 0.
492 if (limit.sequential) {
493 return Duration{0};
494 }

503 template<typename Duration>
504 Duration time_until_key_allows(
505 const LimitData& limit,
506 const KeyState& state,
507 const time_point_t& now
508 ) const {
509 if (limit.sequential &&
510 !state.in_flight_tokens.empty()) {
511 return Duration::max();
512 }
513
514 if (limit.requests_per_period == 0) {
515 return Duration{0};
516 }
517
518 const auto elapsed =
519 std::chrono::duration_cast<Duration>(now - state.start_time);
520
521 const auto period_duration =
522 std::chrono::duration_cast<Duration>(
523 std::chrono::milliseconds(limit.period_ms)
524 );
525
526 if (elapsed >= period_duration ||
527 state.count < limit.requests_per_period) {
528 return Duration{0};
529 }
530
531 return period_duration - elapsed;
532 }

535 void gc_stale_keys(const time_point_t& now) {
536 for (auto limit_it = m_limits.begin(); limit_it != m_limits.end(); ) {
537 auto& limit = limit_it->second;
538 for (auto key_it = limit.keys.begin(); key_it != limit.keys.end(); ) {
539 const auto& state = key_it->second;
540 if (state.in_flight_tokens.empty() && state.count == 0) {
541 key_it = limit.keys.erase(key_it);
542 } else if (state.in_flight_tokens.empty()) {
543 const auto elapsed =
544 std::chrono::duration_cast<std::chrono::milliseconds>(
545 now - state.start_time
546 );
547 if (elapsed.count() >= limit.period_ms) {
548 key_it = limit.keys.erase(key_it);
549 } else {
550 ++key_it;
551 }
552 } else {
553 ++key_it;
554 }
555 }
556
557 if (limit.removed && limit.keys.empty()) {
558 limit_it = m_limits.erase(limit_it);
559 } else {
560 ++limit_it;
561 }
562 }
