48 void cancel_requests() {
49 auto& worker = core::NetworkWorker::get_instance();
50 auto& manager = HttpRequestManager::get_instance();
51 const uint64_t group_id = m_request.group_id;
52 if (worker.is_worker_thread()) {
53 // On the worker thread: execute cancellation inline to avoid self-deadlock.
54 manager.cancel_requests_by_group_id(group_id, nullptr);
55 worker.process();
56 return;
57 }
58 auto promise = std::make_shared<std::promise<void>>();
59 auto future = promise->get_future();
60 manager.cancel_requests_by_group_id(group_id, [promise](){
61 try {
62 promise->set_value();
63 } catch (const std::future_error& e) {
64 if (e.code() == std::make_error_condition(std::future_errc::promise_already_satisfied)) {
65 KURLYK_HANDLE_ERROR(e, "Promise already satisfied in HttpClient::request callback");
66 } else {
67 KURLYK_HANDLE_ERROR(e, "Future error in HttpClient::request callback");
68 }
69 } catch (const std::exception& e) {
70 KURLYK_HANDLE_ERROR(e, "Unhandled exception in HttpClient::request callback");
71 } catch (...) {
72 // Unknown fatal error in request callback
73 }
74 });
75 worker.notify();
76 try {
77 future.get();
78 } catch (const std::exception& e) {
79 KURLYK_HANDLE_ERROR(e, "cancel_requests() future.get() failed");
80 }
81 }

85 void clear_rate_limit(RateLimitType type = RateLimitType::RL_GENERAL) {
86 auto& instance = HttpRequestManager::get_instance();

88 switch (type) {

866 static void safe_set_response(
867 std::shared_ptr<std::promise<HttpResponsePtr>> promise,
868 HttpResponsePtr response) {
869 if (!response || !response->ready) return;
870 try {
871 promise->set_value(std::move(response));
872 } catch (const std::future_error& e) {
873 if (e.code() == std::make_error_condition(std::future_errc::promise_already_satisfied)) {
874 KURLYK_HANDLE_ERROR(e, "Promise already satisfied in HttpClient::request callback");
875 } else {

877 }
878 } catch (const std::exception& e) {

880 } catch (...) {

905 void safe_submit_request(
906 std::shared_ptr<std::promise<HttpResponsePtr>> promise,
907 std::unique_ptr<HttpRequest> request_ptr,
908 HttpResponseCallback callback) {
909 try {
910 const SubmitResult submit_result = submit_request(std::move(request_ptr), std::move(callback));
911 if (!submit_result) {

919 } catch (...) {
